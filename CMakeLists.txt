cmake_minimum_required(VERSION 3.20)
project(Regicide VERSION 0.1.0 LANGUAGES CXX)

add_library(regicide_core src/board.cpp)
target_include_directories(regicide_core PUBLIC include)
target_compile_features(regicide_core PUBLIC cxx_std_20)
function(regicide_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()
regicide_warnings(regicide_core)

add_executable(regicide src/main.cpp)
target_link_libraries(regicide PRIVATE regicide_core)
regicide_warnings(regicide)
include(CTest)
if(BUILD_TESTING)
    add_executable(regicide_tests tests/rules_tests.cpp)
    target_link_libraries(regicide_tests PRIVATE regicide_core)
    regicide_warnings(regicide_tests)
    add_test(NAME rules COMMAND regicide_tests)
endif()
