cmake_minimum_required(VERSION 3.20)
project(Regicide VERSION 0.6.3 LANGUAGES CXX)
find_package(Threads REQUIRED)

add_library(regicide_core src/board.cpp)
target_include_directories(regicide_core PUBLIC include)
target_compile_features(regicide_core PUBLIC cxx_std_20)
function(regicide_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()
regicide_warnings(regicide_core)

add_library(regicide_search src/search.cpp src/evaluation.cpp src/transposition.cpp src/see.cpp src/neural.cpp)
target_link_libraries(regicide_search PUBLIC regicide_core)
regicide_warnings(regicide_search)
add_executable(regicide src/main.cpp src/uci.cpp)
target_link_libraries(regicide PRIVATE regicide_search Threads::Threads)
regicide_warnings(regicide)
add_executable(regicide_profile_evaluation EXCLUDE_FROM_ALL
    scripts/profile_evaluation.cpp tests/reference_evaluation_v054.cpp tests/reference_evaluation_v056.cpp)
target_include_directories(regicide_profile_evaluation PRIVATE tests)
target_link_libraries(regicide_profile_evaluation PRIVATE regicide_search)
regicide_warnings(regicide_profile_evaluation)
add_executable(regicide_evaluation_probe EXCLUDE_FROM_ALL scripts/inspect_evaluation.cpp)
target_link_libraries(regicide_evaluation_probe PRIVATE regicide_search)
regicide_warnings(regicide_evaluation_probe)
include(CTest)
